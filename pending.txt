const DEFAULTS = {
    confirmButtonColor: "#2e53b8",
    cancelButtonColor: "#64748b",
    background: "#ffffff",
    color: "#263247"
};

/* =====================================
   1. NOTIFICACIÓN TOAST
===================================== */

export function notificacion(mensaje, tipo = "success", opciones = {}) {
    return Swal.fire({
        toast: true,
        position: "top-end",
        icon: tipo,
        title: mensaje,

        showConfirmButton: false,
        showCloseButton: true,

        timer: 4000,
        timerProgressBar: true,

        background: DEFAULTS.background,
        color: DEFAULTS.color,

        ...opciones
    });
}

/* =====================================
   2. CONFIRMACIÓN PERSONALIZADA
   Devuelve true si confirma, false si cancela
===================================== */

export async function confirmar(mensaje, opciones = {}) {
    const resultado = await Swal.fire({
        title: "¿Estás seguro?",
        text: mensaje,
        icon: "question",

        showCancelButton: true,
        confirmButtonText: "Sí, continuar",
        cancelButtonText: "Cancelar",

        confirmButtonColor: DEFAULTS.confirmButtonColor,
        cancelButtonColor: DEFAULTS.cancelButtonColor,

        background: DEFAULTS.background,
        color: DEFAULTS.color,

        reverseButtons: true,
        focusCancel: true,

        ...opciones
    });

    return resultado.isConfirmed;
}

/* =====================================
   3. ALERTA MODAL
===================================== */

export function alerta(mensaje, tipo = "info", opciones = {}) {
    return Swal.fire({
        title: "Aviso",
        text: mensaje,
        icon: tipo,

        confirmButtonText: "Aceptar",
        confirmButtonColor: DEFAULTS.confirmButtonColor,

        background: DEFAULTS.background,
        color: DEFAULTS.color,

        ...opciones
    });
}

    a.marca,
    a.modelo,
    a.anio
FROM ambulancia AS a
INNER JOIN estado_ambulancia AS ea
    ON ea.id_estado_ambulancia = a.id_estado_ambulancia
WHERE ea.nombre = 'Activa'
  AND NOT EXISTS (
      SELECT 1
      FROM pendiente AS p
      WHERE p.id_ambulancia = a.id_ambulancia
        AND p.hora_inicio < ?
        AND p.hora_fin > ?
  )
ORDER BY a.matricula;


SELECT
    c.codigo,
    f.nombre,
    f.apellido
FROM conductor AS c
INNER JOIN funcionario AS f
    ON f.id_funcionario = c.id_funcionario
INNER JOIN estado_funcionario AS ef
    ON ef.id_estado_funcionario = f.id_estado_funcionario
WHERE ef.nombre = 'Activo'
  AND c.vencimiento_carnet >= CURDATE()
  AND NOT EXISTS (
      SELECT 1
      FROM pendiente AS p
      WHERE p.codigo_conductor = c.codigo
        AND p.hora_inicio < ?
        AND p.hora_fin > ?
  )
ORDER BY f.apellido, f.nombre;
